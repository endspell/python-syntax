OOP: classes & objects

class Animal:
    count = 0                       # class attribute

    def __init__(self, name):
        self.name = name            # instance attribute
        self._protected = 1         # convention
        self.__private = 2          # name mangled

    def speak(self):                # instance method
        return f"{self.name} makes a sound"

    def __repr__(self):  return f"Animal({self.name!r})"
    def __str__(self):   return self.name

    @classmethod
    def create(cls, name):  return cls(name)

    @staticmethod
    def helper(x):  return x * 2

    @property
    def upper(self):  return self.name.upper()

    @upper.setter
    def upper(self, v):  self.name = v.lower()

a = Animal("Rex")

Iterators, generators & context managers

class Count:
    def __init__(self, n): self.n, self.i = n, 0
    def __iter__(self):  return self
    def __next__(self):
        if self.i >= self.n: raise StopIteration
        self.i += 1
        return self.i

class Vec:
    def __init__(self, x, y): self.x, self.y = x, y
    def __add__(self, o):  return Vec(self.x + o.x, self.y + o.y)
    def __eq__(self, o):   return (self.x, self.y) == (o.x, o.y)
    def __hash__(self):    return hash((self.x, self.y))
    def __len__(self):     return 2
    def __getitem__(self, i):  return (self.x, self.y)[i]
    def __call__(self):    return "called"

Descriptors & __slots__

class Positive:
    def __set_name__(self, owner, name):  self.name = "_" + name
    def __get__(self, obj, objtype=None):
        return self if obj is None else getattr(obj, self.name)
    def __set__(self, obj, value):
        if value <= 0: raise ValueError
        setattr(obj, self.name, value)
    def __delete__(self, obj):  delattr(obj, self.name)

class Item:
    price = Positive()
    __slots__ = ("_price",)       # no per-instance __dict__
